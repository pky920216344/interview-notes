  • 待提交

还是逐行回溯，只是把三套冲突标记压成一个整数的位：cols 管列，ld/rw 管两条对角线，每降一行整体移一位。

思路

  • 与 51 题同款逐行回溯：每行选一列，row == n 时解数加一，只计数不记录棋盘。
  • 三个位掩码 cols / ld / rw 的第 c 位是 1 表示第 c 列被占：本行可选列 avail = 全 1 & ~(cols|ld|rw)。
  • p = avail & (-avail) 取最低可用列；放进下一行时 ld 整体左移一位、rw 整体右移一位——斜线攻击随行下移平移一格。
  • n ≤ 9，一个 int 装得下全部状态；回溯就是 while 里继续取下一个 p。

时间复杂度 O(n!)（剪枝后远小于），空间复杂度 O(n)
1class Solution {2    private int n, count;3 4    public int totalNQueens(int n) {5        this.n = n;6        count = 0;7        dfs(0, 0, 0, 0);8        return count;9    }10 11    // 逐行放；cols/ld/rw 分别是列、左斜线、右斜线已占的列（按位记）12    private void dfs(int row, int cols, int ld, int rw) {13        if (row == n) { count++; return; }14        // 三个方向取并再取反，剩下的就是本行还能放的列15        int avail = ((1 << n) - 1) & ~(cols | ld | rw);16        while (avail != 0) {17            int p = avail & (-avail);   // 取最低一位可用列18            avail ^= p;                 // 从候选去掉，回溯后接着试下一个19            // 下一行：列原样继承，左斜线左移一位，右斜线右移一位20            dfs(row + 1, cols | p, (ld | p) << 1, (rw | p) >> 1);21        }22    }23}
