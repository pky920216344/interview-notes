308. 二维区域和检索 - 矩阵可修改

二维树状数组：单点改值沿 lowbit 链向上修正，前缀和沿链下行累加，区间和用四次前缀和容斥。

思路

  • BIT 的每个结点 tree[i][j] 管住一块矩形（行方向 [i−lowbit+1..i]、列方向 [j−lowbit+1..j]）；两个方向各自独立跳 lowbit。
  • update 是增量语义：先算 val − nums[row][col] 的差值再入树（所以手里要留原矩阵副本）。
  • sumRegion 不直接做：取四次前缀和容斥——大矩形减上、减左、加回左上角重叠。
  • 复杂度：建树 O(mn)、单点更新与区域查询都是 O(log m × log n)；暴力重算是 O(mn) 每查，BIT 是这类「可改可查」场景的标准答案。

1import java.util.*;2 3class NumMatrix {4    private final int[][] tree;   // 1-based 二维树状数组，存的是前缀和5    private final int[][] nums;   // 原矩阵，update 时算差值用6    private final int m, n;7 8    public NumMatrix(int[][] matrix) {9        m = matrix.length;10        n = matrix[0].length;11        tree = new int[m + 1][n + 1];12        nums = matrix;13        for (int i = 0; i < m; i++)14            for (int j = 0; j < n; j++)15                add(i + 1, j + 1, matrix[i][j]);16    }17 18    public void update(int row, int col, int val) {19        // BIT 只支持增量更新，先把这一格的差值算出来20        add(row + 1, col + 1, val - nums[row][col]);21        nums[row][col] = val;22    }23 24    public int sumRegion(int row1, int col1, int row2, int col2) {25        // 容斥：大前缀减去两侧再补回重叠角26        return query(row2 + 1, col2 + 1) - query(row1, col2 + 1)27             - query(row2 + 1, col1) + query(row1, col1);28    }29 30    // (x, y) 单点加上 delta，沿两个方向各自跳 lowbit31    private void add(int x, int y, int delta) {32        for (int i = x; i <= m; i += i & (-i))33            for (int j = y; j <= n; j += j & (-j))34                tree[i][j] += delta;35    }36 37    // (1,1) 到 (x,y) 的前缀和38    private int query(int x, int y) {39        int sum = 0;40        for (int i = x; i > 0; i -= i & (-i))41            for (int j = y; j > 0; j -= j & (-j))42                sum += tree[i][j];43        return sum;44    }45}
