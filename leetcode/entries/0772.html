772. 基本计算器 III

递归下降三层：expr 管加减、term 管乘除、factor 管数字/括号/一元负号；括号里当一个子表达式递归求值。

思路

  • 三层各管一件事：expr 从左到右吃 + -，term 从左到右吃 * /，factor 产出最小单元——乘除天然比加减先算。
  • factor 遇到 (：i++ 跳过它，递归调 expr() 算括号内，再 i++ 吃掉 )，返回值当一个因子。
  • 一元负号在 factor 里处理：读到 - 就返回 -factor()（可叠，如 --3）。
  • 开头 s.replace(" ", "") 去空格；除法用 Java /，向零取整。

最优解

时间复杂度 O(n)，空间复杂度 O(n)（递归深度 = 括号嵌套）
1class Solution {2    private String s;3    private int i;                        // 当前扫到的位置，各层递归共享4 5    public int calculate(String s) {6        this.s = s.replace(" ", "");      // 空格不影响值，先全部去掉7        this.i = 0;8        return expr();9    }10 11    // 加减层：expr := 项 { (+|-) 项 }；乘除已经在 term 里算完了12    private int expr() {13        int v = term();14        while (i < s.length() && (s.charAt(i) == '+' || s.charAt(i) == '-')) {15            char op = s.charAt(i++);16            if (op == '+') v += term();17            else v -= term();18        }19        return v;20    }21 22    // 乘除层：term := 因子 { (*|/) 因子 }，同级从左到右23    private int term() {24        int v = factor();25        while (i < s.length() && (s.charAt(i) == '*' || s.charAt(i) == '/')) {26            char op = s.charAt(i++);27            if (op == '*') v *= factor();28            else v /= factor();29        }30        return v;31    }32 33    // 因子：数字 | ( expr ) | 一元 +/-（如 -(2+3)、-3）34    private int factor() {35        char c = s.charAt(i);36        if (c == '(') {37            i++;                          // 吃掉 '('38            int v = expr();               // 括号里当一个子表达式递归求值39            i++;                          // 吃掉 ')'40            return v;41        }42        if (c == '-' || c == '+') {43            i++;44            return c == '-' ? -factor() : factor();45        }46        int v = 0;47        while (i < s.length() && s.charAt(i) >= '0' && s.charAt(i) <= '9') {48            v = v * 10 + (s.charAt(i++) - '0');   // 完整读多位数字49        }50        return v;51    }52}
