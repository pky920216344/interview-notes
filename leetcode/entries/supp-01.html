补充题1. 排序奇升偶降链表

  • 中等
  • 频度 46
  • 链表
  • 双指针
  • 未经力扣验证

按位置拆成奇位链（升序）和偶位链（降序），偶链反转后也升序，两条归并即整条有序。

思路

  • 奇数位结点本身升序、偶数位本身降序：按「位置」把链表摘成两条链（奇链、偶链），交替改 next 即可。
  • 偶链是降序，反转一次变升序。
  • 两条升序链归并（每次取较小的头）得到整体有序；全程只动指针，O(1) 额外空间。

最优解

时间复杂度 O(n)，空间复杂度 O(1)
1class Solution {2    // 链表奇数位升序、偶数位降序：拆成两条链 → 反转偶链 → 归并3    public ListNode sort(ListNode head) {4        if (head == null || head.next == null) return head;5        ListNode odd = head, even = head.next;   // 奇位链、偶位链的头6        ListNode o = odd, e = even;7        while (e != null && e.next != null) {8            // 交替摘结点：下一个奇位结点进奇链，下一个偶位结点进偶链9            o.next = e.next;10            o = o.next;11            e.next = o.next;12            e = e.next;13        }14        // 偶数长度时奇链尾还挂在偶链尾上，断开15        o.next = null;16        // 偶链反转后也升序，两条归并17        return merge(odd, reverse(even));18    }19 20    // 反转链表：逐个头插到 prev 前面21    private ListNode reverse(ListNode head) {22        ListNode prev = null;23        while (head != null) {24            ListNode next = head.next;25            head.next = prev;26            prev = head;27            head = next;28        }29        return prev;30    }31 32    // 归并两条升序链：每次取较小的头接到 tail 后面33    private ListNode merge(ListNode a, ListNode b) {34        ListNode dummy = new ListNode(0), tail = dummy;35        while (a != null && b != null) {36            if (a.val <= b.val) { tail.next = a; a = a.next; }37            else { tail.next = b; b = b.next; }38            tail = tail.next;39        }40        tail.next = a != null ? a : b;41        return dummy.next;42    }43}
