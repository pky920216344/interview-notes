351. 安卓系统手势解锁

  • 中等
  • 频度 2
  • 回溯
  • 位运算
  • 动态规划
  • 回溯
  • 位掩码
  • 未经力扣验证

DFS + 位掩码：从每个点出发延伸到长度 n，两点一线时必须跨过已连过的中点；长度落在 [m, n] 的每条序列各计 1。

思路

  • 只有 8 组点对要「跨中点」：同行/同列隔一格的 6 组加两条对角线（1-9、3-7 跨 5），用 skip 表查。
  • DFS 从每个起点延伸：下一个点没连过、且（若要跨中点）中点已连过，才能走。
  • 每连一个点 len+1；len ≥ m 时当前序列本身就是一个答案，len == n 就收手。
  • 按起点分头数不会重复：合法序列的首点唯一。回溯用 used[] 数组就行，本题不需要记忆化。

最优解

时间复杂度 O(n!)，空间复杂度 O(n)
1class Solution {2    // 需要「跨过中间点」的点对：skip[a][b] = 中间点（对称填表）3    private int[][] skip = new int[10][10];4    private boolean[] used = new boolean[10];5 6    public int numberOfPatterns(int m, int n) {7        skip[1][3] = skip[3][1] = 2;8        skip[4][6] = skip[6][4] = 5;9        skip[7][9] = skip[9][7] = 8;10        skip[1][7] = skip[7][1] = 4;11        skip[2][8] = skip[8][2] = 5;12        skip[3][9] = skip[9][3] = 6;13        skip[1][9] = skip[9][1] = 5;14        skip[3][7] = skip[7][3] = 5;15        int ans = 0;16        for (int start = 1; start <= 9; start++) {   // 每个合法序列都有唯一的首点，按起点分头数17            used[start] = true;18            ans += dfs(start, 1, m, n);19            used[start] = false;20        }21        return ans;22    }23 24    // cur：当前停在哪个点；len：已用点数。返回从当前状态能延伸出的合法手势总数25    private int dfs(int cur, int len, int m, int n) {26        int cnt = len >= m ? 1 : 0;                  // 当前序列本身就是一个答案27        if (len == n) return cnt;28        for (int next = 1; next <= 9; next++) {29            if (used[next]) continue;30            int mid = skip[cur][next];31            // 两点一线时要跨过的中间点必须已经连过32            if (mid != 0 && !used[mid]) continue;33            used[next] = true;34            cnt += dfs(next, len + 1, m, n);35            used[next] = false;36        }37        return cnt;38    }39}
