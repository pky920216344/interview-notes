每格存现值 + 公式展开表 + 反向依赖表；写入时沿反向依赖递归重算（seen 防环防重复），读取 O(1)。

思路

  • 三个结构：val 现值、fml 公式格 → 展开后的格子列表、rev 格子 → 依赖它的公式格（反向边）。
  • set/sum 改完值后 wave：沿反向边递归重算——公式格先把自己的项递归刷新（后序），再通知依赖自己的格子；seen 集合防环也防同一格重算两遍。
  • sum 里的范围 "A1:B2" 逐格展开，重复出现的项（如 A1 + A1:B2）就加两次；被覆盖写的旧公式要从反向边里摘掉。

最优解

时间复杂度 set/sum O(受影响的依赖链)，get O(1)，空间复杂度 O(格子数 + 依赖边数)
1import java.util.*;2 3class Excel {4    // 格子 id = 行 × 26 + 列（行列都不超过 26，id 唯一）5    private final Map<Integer, Integer> val = new HashMap<>();        // 当前值（普通格、公式格的现值）6    private final Map<Integer, List<Integer>> fml = new HashMap<>();  // 公式格 → 展开后的格子列表7    private final Map<Integer, List<Integer>> rev = new HashMap<>();  // 格子 → 直接依赖它的公式格8 9    public Excel(int height, char width) {10    }11 12    public void set(int row, char column, int v) {13        int x = id(row, column);14        clearFormula(x);          // 覆盖写会顶掉旧公式15        val.put(x, v);16        wave(x);                  // 依赖它的公式格沿链重算17    }18 19    public int get(int row, char column) {20        return val.getOrDefault(id(row, column), 0);21    }22 23    public int sum(int row, char column, List<String> numbers) {24        int x = id(row, column);25        clearFormula(x);26        List<Integer> terms = new ArrayList<>();27        for (String s : numbers)28            for (int[] cell : expand(s))29                terms.add(cell[0] * 26 + cell[1]);30        fml.put(x, terms);31        for (int t : terms) rev.computeIfAbsent(t, k -> new ArrayList<>()).add(x);32        wave(x);33        return val.get(x);34    }35 36    // x 的值变了：递归重算所有直接或间接依赖它的公式格；seen 防环、防重复算37    private void wave(int x) {38        recompute(x, new HashSet<>());39    }40 41    // 把公式格 x 的值重算对：它的项可能也过期，先递归重算项，再通知依赖 x 的格子42    private void recompute(int x, Set<Integer> seen) {43        if (!seen.add(x)) return;44        if (fml.containsKey(x)) {45            int sum = 0;46            for (int t : fml.get(x)) {47                if (fml.containsKey(t)) recompute(t, seen);48                sum += val.getOrDefault(t, 0);49            }50            val.put(x, sum);51        }52        for (int d : rev.getOrDefault(x, List.of())) recompute(d, seen);53    }54 55    // 撤掉 x 上的旧公式：清掉展开表，并从各项的反向依赖里摘除 x56    private void clearFormula(int x) {57        List<Integer> old = fml.remove(x);58        if (old != null)59            for (int t : old) {60                List<Integer> l = rev.get(t);61                if (l != null) l.remove(Integer.valueOf(x));62            }63    }64 65    // "A1" 是单格，"A1:B2" 是矩形范围（行、列都按小到大展开）66    private List<int[]> expand(String s) {67        int colon = s.indexOf(':');68        int[] a = parse(colon < 0 ? s : s.substring(0, colon));69        if (colon < 0) return List.of(a);70        int[] b = parse(s.substring(colon + 1));71        List<int[]> out = new ArrayList<>();72        for (int r = Math.min(a[0], b[0]); r <= Math.max(a[0], b[0]); r++)73            for (int c = Math.min(a[1], b[1]); c <= Math.max(a[1], b[1]); c++)74                out.add(new int[]{r, c});75        return out;76    }77 78    // "B3" → 行 3、列 1（B − A）79    private int[] parse(String s) {80        return new int[]{Integer.parseInt(s.substring(1)), s.charAt(0) - 'A'};81    }82 83    private int id(int row, char column) {84        return row * 26 + (column - 'A');85    }86}
