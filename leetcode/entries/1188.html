1188. 设计有限阻塞队列

  • 中等
  • 频度 1
  • 并发
  • 设计
  • 多线程
  • 未经力扣验证

循环数组 + 一把 ReentrantLock、两个 Condition：队满生产者等 notFull，队空消费者等 notEmpty，操作完唤醒对方一个。

思路

  • 循环数组存元素：head 指向下一个出队位置，tail 指向下一个写入位置，下标用 % capacity 回绕。
  • enqueue：满了就在 notFull 上 await；写入后 signal 一个 notEmpty（只腾出一个元素，唤醒一个消费者就够）。
  • dequeue：空了就在 notEmpty 上 await；取走后 signal 一个 notFull。await 必须放在 while 里：被唤醒后条件可能又被其他线程抢走。
  • 所有读写都在同一把锁里，size() 也要持锁读，才能保证可见性和一致性。

1import java.util.concurrent.locks.Condition;2import java.util.concurrent.locks.ReentrantLock;3 4class BoundedBlockingQueue {5    private final int[] buf;   // 循环数组：head 读、tail 写6    private int head, tail, size;7    private final ReentrantLock lock = new ReentrantLock();8    private final Condition notFull = lock.newCondition();9    private final Condition notEmpty = lock.newCondition();10 11    public BoundedBlockingQueue(int capacity) {12        buf = new int[capacity];13    }14 15    public void enqueue(int element) throws InterruptedException {16        lock.lock();17        try {18            // 用 while 而不是 if：被唤醒后条件可能又被别人抢走19            while (size == buf.length) notFull.await();20            buf[tail] = element;21            tail = (tail + 1) % buf.length;22            size++;23            notEmpty.signal();   // 腾出了一个元素，唤醒一个消费者即可24        } finally {25            lock.unlock();26        }27    }28 29    public int dequeue() throws InterruptedException {30        lock.lock();31        try {32            while (size == 0) notEmpty.await();33            int v = buf[head];34            head = (head + 1) % buf.length;35            size--;36            notFull.signal();   // 腾出了一个空位，唤醒一个生产者即可37            return v;38        } finally {39            lock.unlock();40        }41    }42 43    public int size() {44        lock.lock();45        try {46            return size;47        } finally {48            lock.unlock();49        }50    }51}
