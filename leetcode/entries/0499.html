还是「滚到撞墙」的停点图跑 Dijkstra；堆按（滚动距离, 方向串）排序，距离并列时字典序小的路线先弹；滚动途中扫过洞口直接掉进去。

思路

  • 与 505 同款滚动建模：从停点沿一个方向 while 滚到撞墙，边权 = 滚过的格数；不同的是滚动途中扫过洞口就停（fell），终点是洞而不是停点。
  • 最短距离用 Dijkstra；距离并列要输出字典序最小的方向串：堆元素带方向串，按 (dist, 串) 排序——边权 ≥ 1 保证首次弹出即定稿。
  • 松弛条件是二元组比较：dist 更小才更新；dist 相等时新串字典序更小才更新。
  • 方向串按 d、l、r、u 拼接；堆空还没弹到洞就是到不了，返回 "impossible"。

1import java.util.*;2 3class Solution {4    public String findShortestWay(int[][] maze, int[] ball, int[] hole) {5        int m = maze.length, n = maze[0].length;6        int[][] dist = new int[m][n];7        for (int[] row : dist) Arrays.fill(row, Integer.MAX_VALUE);8        String[][] path = new String[m][n];9        dist[ball[0]][ball[1]] = 0;10        path[ball[0]][ball[1]] = "";11        boolean[][] done = new boolean[m][n];12        // 堆按 (滚动距离, 方向串) 排序：距离并列时，字典序小的路线先弹13        PriorityQueue<Node> pq = new PriorityQueue<>((a, b) -> a.d != b.d ? a.d - b.d : a.s.compareTo(b.s));14        pq.add(new Node(0, ball[0], ball[1], ""));15        // d l r u 四个方向；滚到洞口算到站，撞墙算停点16        int[][] dirs = {{1, 0}, {0, -1}, {0, 1}, {-1, 0}};17        String[] ch = {"d", "l", "r", "u"};18        while (!pq.isEmpty()) {19            Node cur = pq.poll();20            int r = cur.r, c = cur.c;21            if (done[r][c]) continue;   // 边权 ≥ 1，首次弹出即定稿22            done[r][c] = true;23            if (r == hole[0] && c == hole[1]) return cur.s;24            for (int k = 0; k < 4; k++) {25                int nr = r, nc = c, len = 0;26                boolean fell = false;27                while (open(maze, nr + dirs[k][0], nc + dirs[k][1])) {28                    nr += dirs[k][0];29                    nc += dirs[k][1];30                    len++;31                    if (nr == hole[0] && nc == hole[1]) { fell = true; break; }   // 滚过洞口就掉进去32                }33                if (len == 0) continue;34                int nd = cur.d + len;35                int tr = fell ? hole[0] : nr, tc = fell ? hole[1] : nc;36                if (done[tr][tc]) continue;37                String ns = cur.s + ch[k];38                // 距离更短才更优；距离并列时比方向串的字典序39                if (nd < dist[tr][tc] || (nd == dist[tr][tc] && ns.compareTo(path[tr][tc]) < 0)) {40                    dist[tr][tc] = nd;41                    path[tr][tc] = ns;42                    pq.add(new Node(nd, tr, tc, ns));43                }44            }45        }46        return "impossible";47    }48 49    private boolean open(int[][] maze, int r, int c) {50        return r >= 0 && r < maze.length && c >= 0 && c < maze[0].length && maze[r][c] == 0;51    }52 53    private static class Node {54        int d, r, c;55        String s;56 57        Node(int d, int r, int c, String s) {58            this.d = d;59            this.r = r;60            this.c = c;61            this.s = s;62        }63    }64}
